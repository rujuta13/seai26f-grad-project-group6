Use the Python 3.13 standard library plus the ezr library, importable as `import ezr` (its full source is below). No other third-party packages.

```python
# ezr.py
#!/usr/bin/env python3 -B
"""
ezr.py: multi-objective XAI using minimal labels.   
Faster, simpler AI. Better maps, not bigger boots.    
(c) 2026 Tim Menzies <timm@ieee.org> MIT license.    

Options:    
      -Start=4         acquire: initial random labels   
      -Stop=50         acquire: total labelling budget   
      -Few=128         holdout: max training rows   
      -Cuts=8          tree: splits to try, per num   
      -Leaf=4          tree: min rows in any leaf  
      -Check=5         holdout: top picks to label  
      -Repeats=20      holdout: how many train/test splits  
      -Seed=1234567891 random number seed  
      -File=~/gits/moot/optimize/misc/auto93.csv   

One file, five layers.  Each calls the layers above it and
knows nothing of the ones below.  Cut on the banners to get
five files back:    

      core     columns, tables, distance.  Calls nothing.    
      acquire  spend a label budget.    
      tree     recursive splits, and a way to read them.    
      rig      wins, holdout.  Takes its ranker as an argument,    
               so the tree is its default, not its requirement.    
      stats    cliffs, ks, cohen.  A leaf: nothing calls it.    

"""
# pylint: disable=bad-indentation,multiple-statements  
# pylint: disable=invalid-name,ungrouped-imports   
# pylint: disable=attribute-defined-outside-init   
# pylint: disable=protected-access,broad-exception-caught  
from typing import Any
import os, re, sys, random, traceback # pylint: disable=C0410
from math import exp, log2, sqrt
from collections.abc import Callable, Iterable, Iterator

#-- types --------------------------------------------------
type QTY    = int | float             # any number
type ATOM   = QTY | str               # one cell; "?" = unknown
type ROW    = list[ATOM]
type MIDS   = dict[int, ATOM]         # a centroid
type ROWS   = list[ROW]
type NUMS   = list[QTY]
type NUM    = Num                     # classes, defined below
type SYM    = Sym
type COL    = NUM | SYM               # sumamry of a column
type TBL    = o                       # rows and columns
type NODE   = o                       # one node of a tree
type GO     = Callable[[ROW], bool]   # which way does a row go?
type ORACLE = Callable[[ROW], ROW]    # labels one row
type PICKER = Callable[[TBL, int, ORACLE], ROWS]
type RANKER = Callable[[TBL], Callable[[ROW], float]]
                                      # ranks unlabelled rows

#-- misc ---------------------------------------------------
def say(x: Any, p: int = 2) -> str:
  "Floats get P decimals; dicts list their keys."
  if isinstance(x,float): x= f"{x:.{p}f}".rstrip("0").rstrip(".")
  if isinstance(x, dict):
    x= "{"+", ".join(f"{k}: {say(v,p)}" for k,v in x.items())+"}"
  return str(x)

class o(dict):
  "A dict you can poke with a dot."
  __getattr__, __setattr__ = dict.__getitem__, dict.__setitem__
  __repr__ = say

def atom(s: str) -> ATOM:
  "'22' -> 22. '3.1' -> 3.1. 'True' -> True. 'x' -> 'x'."
  for fn in (int, float):
    try: return fn(s)
    except ValueError: pass
  s = s.strip()
  return {"True": True, "False": False}.get(s, s)

def csv(file: str) -> Iterator[ROW]:
  "ROWS of FILE, each cell coerced."
  with open(os.path.expanduser(file),  # -sig drops any BOM
           encoding="utf-8-sig") as f:
    for line in f:
      if line.strip(): yield [atom(s) for s in line.split(",")]

#-- columns ------------------------------------------------
def Col(txt: str = " ", at: int = 0) -> COL:
  "Factory that returns NUM or SYM."
  return Num(txt,at) if txt[0].isupper() else Sym(txt,at)

class Num:
  "Place to summarize stream of numbers."
  __slots__ = ("at", "txt", "n", "mu", "m2", "sd", "goal")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n = at, txt, 0
    self.mu = self.m2 = self.sd = 0
    self.goal = 0 if txt[-1] == "-" else 1
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

class Sym:
  "Place to summarize stream of Symbols.  The class is the tag."
  __slots__ = ("at", "txt", "n", "has")
  def __init__(self, txt: str = " ", at: int = 0) -> None:
    self.at, self.txt, self.n, self.has = at, txt, 0, {}
  __repr__ = lambda self: say({k: getattr(self,k)
                               for k in self.__slots__})

def add(col: COL, v: ATOM, inc: int = 1) -> None:
  "Show V to COL.  INC=-1 takes it away.  `?` changes nothing."
  if v != "?":
    col.n += inc
    if type(col) is Sym:
      col.has[v] = col.has.get(v, 0) + inc
    else:
      d = v - col.mu                      # Welford, for `div`
      col.mu += inc * d / max(1, col.n)
      col.m2  = max(0, col.m2 + inc * d * (v - col.mu))
      col.sd  = 0 if col.n < 2 else (col.m2/(col.n-1))**.5

def mid(col: COL) -> ATOM:
  "Middle: the mean, or the most common symbol; ? if none."
  return (max(col.has, key=col.has.get) if col.has else "?"
          ) if type(col) is Sym else col.mu

def mids(tbl: TBL) -> MIDS:
  "Every column's middle, cached until the next addCols."
  tbl.mids = tbl.mids or {at: mid(c) for at,c in tbl.cols.items()}
  return tbl.mids

def div(col: COL) -> float:
  "Spread: entropy, or standard deviation."
  return (-sum(v/col.n*log2(v/col.n) for v in col.has.values())
          if type(col) is Sym else col.sd)  # sd kept by `add`

def norm(col: COL, v: ATOM) -> ATOM:
  "To 0..1, by the logistic curve.  Symbols do not scale."
  if type(col) is Sym: return v
  z = max(-3, min(3, (v - col.mu) / (1e-32 + div(col))))
  return 1 / (1 + exp(-1.7 * z))

#-- tables -------------------------------------------------
def Tbl(src: Iterable[ROW]) -> TBL:
  "Header names the columns: X skip, +- goal, ! klass."
  src = iter(src)
  tbl = Cols(o(rows=[], cols={}, x=[], y=[], mids=None,
               names=next(src)))
  for row in src: addRow(tbl, row)
  return tbl

def Cols(tbl: TBL) -> TBL:
  "Create column roles."
  for at, s in enumerate(tbl.names):
    if s[-1] == "X": continue            # skip me entirely
    tbl.cols[at] = Col(s,at)
    (tbl.y if s[-1] in "+-!" else tbl.x).append(tbl.cols[at])
  return tbl

def addRow(tbl: TBL, row: ROW) -> ROW:
  "Keep ROW, and show it to my columns."
  tbl.rows += [row]
  return addCols(tbl, row)

def addCols(tbl: TBL, row: ROW, inc: int = 1) -> ROW:
  "Show ROW to my columns, without keeping it."
  tbl.mids = None                 # a new row moves the middles
  for at, col in tbl.cols.items(): add(col, row[at], inc)
  return row

def clone(tbl: TBL, rows: ROWS = None) -> TBL:
  "An empty copy of TBL, plus ROWS."
  return Tbl([tbl.names] + (rows or []))

#-- distance -----------------------------------------------
def dist(vs: Iterable[QTY], n: int) -> float:
  "Root mean square of N gaps."
  return sqrt(sum(v*v for v in vs) / n)

def ydist(tbl: TBL, row: ROW) -> float:
  "How far ROW's goals are from the best they could be."
  return dist((abs(norm(c, row[c.at]) - c.goal)
                    for c in tbl.y), len(tbl.y))

def gap(col: COL, a: ATOM, b: ATOM) -> QTY:
  "Distance between two values of one column."
  if a == "?" or b == "?": return 1    # unknown = far
  return a!=b if type(col) is Sym else abs(norm(col,a)-norm(col,b))

def xdist(tbl: TBL, r1: ROW, r2: ROW | MIDS) -> float:
  "How far apart two rows are, over the x columns."
  return dist((gap(c, r1[c.at], r2[c.at])
                    for c in tbl.x), len(tbl.x))

#-- acquire ------------------------------------------------

def oracle(row: ROW) -> ROW:
  """Labeller. Rows here arrive with ys, so nothing to do.
  Real oracles fill in any missing y cells (and can close
  over a TBL to know which cells those are)."""
  return row

def acquire(tbl: TBL, cap: int | None = None,
            label: ORACLE = oracle) -> ROWS:
  "Label near the best rows, far from the rest."
  cap  = cap or the.Stop
  todo = random.sample(tbl.rows, len(tbl.rows))[:the.Few]
  both = clone(tbl, [label(todo.pop())
                     for _ in range(the.Start)])
  both.rows.sort(key=lambda r: ydist(both, r))
  n    = int(sqrt(the.Start))
  best,rest = clone(tbl,both.rows[:n]), clone(tbl,both.rows[n:])
  while todo and len(both.rows) < cap:
    cb, cr = mids(best), mids(rest)    # two centroids
    todo.sort(key=lambda z: xdist(tbl,z,cr) - xdist(tbl,z,cb))
    addRow(best, addRow(both, label(todo.pop())))
    best.rows.sort(key=lambda z: ydist(both, z))  # no y leak
    if len(best.rows) > sqrt(len(both.rows)): # best stays small
      addRow(rest, addCols(best, best.rows.pop(), -1))
  return both.rows

def grabs(tbl: TBL, cap: int | None = None,
          label: ORACLE = oracle) -> ROWS:
  "The straw man: same budget, taken at random."
  cap = cap or the.Stop
  return [label(r) for r in
          random.sample(tbl.rows, min(cap, len(tbl.rows)))]

#-- tree ---------------------------------------------------
# Calls core.  Nothing above this line calls down here.
# NODE = o(rows, at, v, go, kids, mu).
def cut(tbl: TBL, rows: ROWS) -> tuple[int,ATOM,GO]|None:
  "The (at, v, go) whose two sides have the tightest ys."
  ys = [ydist(tbl, r) for r in rows]
  out, least = None, 1e30
  for col in tbl.x:
    for v, go in candidates(col, rows):
      a, b = Num(), Num()   # one pass, so one `go` per row
      for r,y in zip(rows,ys): add(a if go(r) else b, y)
      if a.n >= the.Leaf and b.n >= the.Leaf:
        score = (div(a)*a.n + div(b)*b.n) / (a.n + b.n)
        if score < least: out, least = (col.at, v, go), score
  return out

def candidates(col: COL, rows: ROWS) -> Iterator[tuple[ATOM,GO]]:
  "What splits to try: one per symbol, or CUTS per num."
  at = col.at
  if type(col) is Sym:
    for v in sorted({r[at] for r in rows if r[at] != "?"}):
      yield v, lambda r,v=v,at=at: r[at] == v
  else:                # by rank, not value: THESE rows are dense
    xs = sorted(r[at] for r in rows if r[at] != "?")
    k  = max(1, len(xs)//the.Cuts)     # cuts, spaced by rank
    for v in xs[k-1:-1:k]:
      yield v, (lambda r,v=v,at=at: r[at] != "?" and r[at] <= v)

def tree(tbl: TBL, rows: ROWS) -> NODE:
  "Split while a split keeps both sides big enough."
  node = o(rows=rows, at=None, v=None, go=None, kids=[],
           mu=sum(ydist(tbl,r) for r in rows)/len(rows))
  if len(rows) > the.Leaf and (found := cut(tbl, rows)):
    at, v, go = found
    yes, no = [], []
    for r in rows: (yes if go(r) else no).append(r)
    node.at, node.v, node.go = at, v, go # cut kepts sides big
    node.kids = [tree(tbl, yes), tree(tbl, no)]
  return node

def leaf(node: NODE, row: ROW) -> NODE:
  "Walk ROW down to its leaf."
  while node.kids: node = node.kids[0 if node.go(row) else 1]
  return node

def leafs(node: NODE) -> list[NODE]:
  "Return leaves of this tree."
  return [x for k in node.kids for x in leafs(k)] or [node]

def show(tbl: TBL, node: NODE, pre: str | None = None,
         edge: str = "") -> None:
  "Tree on the left; ydist and n on the right."
  ls = sorted(leafs(node), key=lambda z: z.mu)
  def walk(z: NODE, pre: str | None, edge: str) -> None:
    m = "+" if z is ls[0] else "-" if z is ls[-1] else " "
    print(f"{m} {round(100*z.mu):>4} {len(z.rows):>5}   "
          f"{(pre or '')+edge}".rstrip())
    op = "=" if type(tbl.cols[z.at or 0]) is Sym else "<="
    no = "!=" if op == "=" else ">"
    for kid, oper in zip(z.kids, [op, no]):
      walk(kid, "" if pre is None else pre + "|  ",
           f"{tbl.names[z.at]} {oper} {say(z.v)}")
  print(" ydist     n")
  walk(node, pre, edge)



#-- rig ----------------------------------------------------
# Calls core, acquire, and one RANKER -- tree's, by default.
def ranker(lab: TBL) -> Callable[[ROW], float]:
  "Grow a tree on LAB; score any row by its leaf's mean ydist."
  node = tree(lab, lab.rows)
  return lambda r: leaf(node, r).mu

def wins(tbl: TBL) -> Callable[[ROW], float]:
  "100 at the best row, 0 at an average one."
  ys = sorted(ydist(tbl, r) for r in tbl.rows)
  lo, avg = ys[0], sum(ys)/len(ys)
  return lambda r: max(-100, min(100, 100*(1 - (ydist(tbl,r)-lo)
                                            / (avg-lo+1e-32))))

def holdout(tbl: TBL, pick: PICKER = acquire,
            rank: RANKER = ranker,
            label: ORACLE = oracle) -> ROW:
  "Train on half; of CHECK guesses on the rest, pick best."
  rows  = random.sample(tbl.rows, len(tbl.rows))
  n     = len(rows)//2
  tr    = clone(tbl, rows[:n][:the.Few])
  lab   = clone(tbl, pick(tr, the.Stop - the.Check, label))
  top   = sorted(rows[n:], key=rank(lab))
  return min(top[:the.Check], key=lambda r: ydist(lab, label(r)))


#-- stats --------------------------------------------------
# A leaf: it calls core, and nothing here calls it.
def adds(vs: Iterable[ATOM], col: COL | None = None) -> COL:
  "Every item of VS into COL."
  col = Num() if col is None else col
  for v in vs: add(col, v)
  return col

def cohen(xs: NUMS, ys: NUMS, d: float = 0.35,
          eps: float = 0) -> bool:
  "Are the means within D pooled standard deviations?"
  a, b = adds(xs), adds(ys)
  pool = ((a.n-1)*div(a)**2 + (b.n-1)*div(b)**2)/(a.n+b.n-2)
  return abs(a.mu - b.mu) <= max(eps, d * sqrt(pool))

def cliffs(xs: NUMS, ys: NUMS, d: float = 0.197) -> bool:
  "Sorted in.  Is the rank imbalance small enough?"
  gt = lt = j = k = 0
  for x in xs:
    while j < len(ys) and ys[j] <  x: j += 1; k = j
    while k < len(ys) and ys[k] <= x: k += 1
    gt += j; lt += len(ys) - k
  return abs(gt - lt) / (len(xs) * len(ys)) <= d

def ks(xs: NUMS, ys: NUMS, a: float = 1.36) -> bool:
  "Sorted in.  95% Kolmogorov-Smirnov."
  n, m, i, j, d = len(xs), len(ys), 0, 0, 0
  while i < n and j < m:
    v = min(xs[i], ys[j])
    while i < n and xs[i] <= v: i += 1
    while j < m and ys[j] <= v: j += 1
    d = max(d, abs(i/n - j/m))
  return d <= a * sqrt((n + m) / (n * m))

def same(xs: NUMS, ys: NUMS, eps: float = 0) -> bool:
  """Indistinguishable.  Always cliffs and ks; EPS adds cohen, and
  doubles as its effect size (eps=0 skips that third test)."""
  xs, ys = sorted(xs), sorted(ys)
  return (cliffs(xs,ys) and ks(xs,ys)
          and (not eps or cohen(xs,ys,d=eps)))

#-- start --------------------------------------------------
the = o(_defaults=o())
for _k, _v in re.findall(r"(\w+)=(\S+)", __doc__ or ""):
  the[_k] = the._defaults[_k] = atom(_v)

def run(f: Callable) -> int:
  "Run F. Demos may poke `the`, so always put it back."
  random.seed(the.Seed)
  try:                   f()
  except Exception as e: traceback.print_exception(e); return 1
  finally:               the.update(the._defaults)
  return 0

def main(args: list[str], funcs: dict) -> None:
  "Dispatch --x to FUNCS' eg_x; -key val updates `the`."
  while args:
    s = args.pop(0)
    if   s[:2] == "--":
      run(funcs.get("eg_" + s[2:], funcs["eg_h"]))
    elif s[1:] in the : the[s[1:]] = atom(args.pop(0))
    else: print(f"unknown arg: {s}")
```

Data conventions:
- Data files are csv files (MOOT format). Column names ending in "+" are goals to maximise,
  "-" goals to minimise; names ending in "X" are ignored; every other column is an input.
- A *row* is a dict {column name: value}; numbers are floats, other cells are strings.
- Row indices count data rows of a csv from 0 (header excluded).
- `train_csv` / `test_csv` are paths to such files, about 800 / 200 rows.

```python
# step1.py
"Step 1: families (clusters of flag sets) and best (unusually good rows)."
import random
import ezr

K     = 5             # number of families to find
ITERS = 20            # cap on k-modes iterations
SEED  = 1234567891    # fixed seed, so results repeat
FRAC  = 0.05          # "best" is at most this fraction of rows

def _load(train_csv: str):
  "Table of TRAIN_CSV, with columns and goals from its header."
  return ezr.Tbl(ezr.csv(train_csv))

def families(train_csv: str) -> list[int]:
  "One family (cluster) label per data row of TRAIN_CSV, in file order."
  tbl  = _load(train_csv)
  rows = tbl.rows
  if not rows: return []
  k     = min(K, len(rows))
  rng   = random.Random(SEED)
  cents = [list(r) for r in rng.sample(rows, k)]   # seed centroids with rows
  labels = [-1] * len(rows)
  for _ in range(ITERS):
    # assign each row to its nearest centroid, over the input (x) columns
    new = [min(range(k), key=lambda j, r=r: ezr.xdist(tbl, r, cents[j]))
           for r in rows]
    if new == labels: break
    labels = new
    # move each centroid to the middle (mode per symbol) of its members
    for j in range(k):
      members = [r for r, lab in zip(rows, labels) if lab == j]
      if members:
        cents[j] = ezr.mids(ezr.clone(tbl, members))
  return labels

def best(train_csv: str) -> list[int]:
  "Row indices of the unusually good rows (best on all goals together), at most 5% of rows, best first."
  tbl   = _load(train_csv)
  rows  = tbl.rows
  n     = int(FRAC * len(rows))
  order = sorted(range(len(rows)), key=lambda i: ezr.ydist(tbl, rows[i]))
  return order[:n]
```

### step2.py
```python
# step2.py
"Step 2: guess each goal of a flag set from its nearest training rows."
import ezr

K  = 20     # nearest training rows to consult
LO = 0.10   # low end of the "fairly sure" range
HI = 0.90   # high end of the "fairly sure" range

_CACHE: dict = {}

def _load(train_csv: str):
  "Table of TRAIN_CSV, cached so repeated guesses are cheap."
  if train_csv not in _CACHE:
    _CACHE[train_csv] = ezr.Tbl(ezr.csv(train_csv))
  return _CACHE[train_csv]

def _pct(xs: list, p: float) -> float:
  "Percentile P (0..1) of sorted XS, by linear interpolation."
  i  = p * (len(xs) - 1)
  lo = int(i)
  hi = min(lo + 1, len(xs) - 1)
  return xs[lo] + (xs[hi] - xs[lo]) * (i - lo)

def _coerce(v):
  "Strings become atoms; other values pass through."
  return ezr.atom(v) if isinstance(v, str) else v

def _full(tbl, row) -> list:
  """Lift ROW (goal columns removed) into a full-width row.
  Goal cells, and any cells not given, are '?'."""
  names = tbl.names
  out   = ["?"] * len(names)
  if isinstance(row, dict):                  # {column name: value}
    for c in tbl.x:
      if c.txt in row: out[c.at] = _coerce(row[c.txt])
  else:                                      # list, in header order, goals removed
    keep = [i for i, n in enumerate(names) if n[-1] not in "+-!"]
    for i, v in zip(keep, row): out[i] = _coerce(v)
  return out

def guess(train_csv, row) -> dict[str, tuple[float, float, float]]:
  """Return {goal: (estimate, low, high)} for every goal.
  The estimate is the median goal value of the K training rows
  nearest to ROW over the input columns; low..high is the
  LO..HI percentile band of those same neighbours."""
  tbl  = _load(train_csv)
  q    = _full(tbl, row)
  rows = tbl.rows
  near = sorted(rows, key=lambda r: ezr.xdist(tbl, q, r))[:K]
  out  = {}
  for c in tbl.y:
    xs = sorted(r[c.at] for r in near if r[c.at] != "?")
    if not xs:                               # no neighbour has this goal
      xs = sorted(r[c.at] for r in rows if r[c.at] != "?")
    if not xs:
      out[c.txt] = (0.0, 0.0, 0.0)
      continue
    lo, est, hi = _pct(xs, LO), _pct(xs, 0.5), _pct(xs, HI)
    out[c.txt] = (float(est), float(lo), float(hi))
  return out
```

```python
# step3.py
"Step 3: certify whether two sets of run times really differ."
import ezr

def same(xs, ys) -> bool:
  """True if XS and YS are NOT really different (the gap is noise),
  False if the difference is real. Both ezr tests (Cliff's delta and
  Kolmogorov-Smirnov) must find no difference for this to return True."""
  xs = [float(x) for x in xs]
  ys = [float(y) for y in ys]
  if not xs or not ys:
    raise ValueError("same: need at least one number in each sample")
  return bool(ezr.same(xs, ys))
```

### step4.py
```python
# step4.py
"Step 4: flow. Keep good rows as they stream in; warn when they drift from train."
import random
import ezr

QUANTILE = 0.25   # a row is "worth keeping" if its goal distance is in train's best 25%
WINDOW   = 15     # recent rows compared against the train baseline, for drift
BASE_N   = 60     # train rows sampled to build the drift baseline
SEED     = 1234567891

_CACHE: dict = {}

def _coerce(v):
  "Strings become atoms; other values pass through."
  return ezr.atom(v) if isinstance(v, str) else v

def _full(tbl, row) -> list:
  """Lift ROW into a full-width row. Accepts a {column name: value} dict,
  a full-width list, or a list with goal columns removed (header order).
  Any cell not given is '?'."""
  names = tbl.names
  if isinstance(row, dict):
    out = ["?"] * len(names)
    for at, n in enumerate(names):
      if n in row: out[at] = _coerce(row[n])
    return out
  row = list(row)
  if len(row) == len(names):
    return [_coerce(v) for v in row]
  out  = ["?"] * len(names)
  keep = [i for i, n in enumerate(names) if n[-1] not in "+-!"]
  for i, v in zip(keep, row): out[i] = _coerce(v)
  return out

def _ydist(tbl, row):
  "Distance of ROW's numeric goals from their best; None if no goal is known."
  sq, n = 0.0, 0
  for c in tbl.y:
    if type(c) is not ezr.Num: continue
    v = row[c.at]
    if v == "?": continue
    sq += (ezr.norm(c, v) - c.goal) ** 2
    n  += 1
  return (sq / n) ** .5 if n else None

def _nn(tbl, row, skip: int | None = None) -> float:
  "Distance from ROW to its nearest train row (ignoring train row SKIP)."
  return min((ezr.xdist(tbl, row, t)
              for j, t in enumerate(tbl.rows) if j != skip),
             default=1.0)

def _load(train_csv: str):
  "Train table, keep-threshold, and drift baseline; cached per file."
  if train_csv not in _CACHE:
    tbl = ezr.Tbl(ezr.csv(train_csv))
    ys  = sorted(d for d in (_ydist(tbl, r) for r in tbl.rows)
                 if d is not None)
    thr = ys[int(QUANTILE * (len(ys) - 1))] if ys else float("inf")
    rng  = random.Random(SEED)
    pick = rng.sample(range(len(tbl.rows)), min(BASE_N, len(tbl.rows)))
    base = [_nn(tbl, tbl.rows[i], skip=i) for i in pick]
    _CACHE[train_csv] = (tbl, thr, base)
  return _CACHE[train_csv]

def stream(train_csv, rows) -> list[bool]:
  """One keep decision per row, in order. A row is kept if its goals are
  among the best QUANTILE of the train rows. Decisions use only TRAIN_CSV
  and the row at hand."""
  tbl, thr, _ = _load(train_csv)
  out = []
  for r in rows:
    d = _ydist(tbl, _full(tbl, r))
    out.append(d is not None and d <= thr)
  return out

def drift(train_csv, rows) -> int | None:
  """Index into ROWS of the first row at which the last WINDOW rows no
  longer look like train (by nearest-train distance over the x columns),
  or None if they never do."""
  tbl, _, base = _load(train_csv)
  recent: list[float] = []
  for i, r in enumerate(rows):
    recent.append(_nn(tbl, _full(tbl, r)))
    if len(recent) > WINDOW: recent.pop(0)
    if len(recent) == WINDOW and not ezr.same(base, recent):
      return i
  return None
```

```python
import ezr

MIN  = 2     # smallest group (rows) a split may isolate
CUTS = 4     # thresholds tried per numeric column

def _ydist(tbl, row):
  "Distance of ROW's numeric goals from their best; None if no goal is known."
  sq, n = 0.0, 0
  for c in tbl.y:
    if type(c) is not ezr.Num: continue
    v = row[c.at]
    if v == "?": continue
    sq += (ezr.norm(c, v) - c.goal) ** 2
    n  += 1
  return (sq / n) ** .5 if n else None

def _splits(col, xs: list):
  "Yield (label, test) pairs: one per symbol value, or CUTS thresholds per number."
  if type(col) is ezr.Sym:
    for v in sorted(set(xs), key=str):
      yield v, (lambda x, v=v: x == v)
  else:
    ys = sorted(xs)
    for i in range(1, CUTS + 1):
      t = ys[(i * (len(ys) - 1)) // (CUTS + 1)]
      yield t, (lambda x, t=t: x <= t)
      yield t, (lambda x, t=t: x > t)

def blame(train_csv) -> list[str]:
  "Input column names, most to blame for bad rows (bad on the goals) first."
  tbl  = ezr.Tbl(ezr.csv(train_csv))
  seen = []
  for r in tbl.rows:
    d = _ydist(tbl, r)
    if d is not None: seen.append((r, d))
  score = {}
  for col in tbl.x:
    vals = [(r[col.at], d) for r, d in seen if r[col.at] != "?"]
    best = 0.0
    if vals:
      for _, test in _splits(col, [x for x, _ in vals]):
        yes = [d for x, d in vals if test(x)]
        no  = [d for x, d in vals if not test(x)]
        if len(yes) < MIN or len(no) < MIN: continue
        gap = sum(yes) / len(yes) - sum(no) / len(no)
        best = max(best, gap * len(yes) ** .5)
    score[col.txt] = best
  return sorted(score, key=lambda k: -score[k])
```

```python
# step6.py
"Step 6: justify. Plain reasons for good builds, and a one-sentence verdict per build."
import ezr

QUANTILE = 0.25   # a build is "good" if its goals are in the best 25% of train
TOP      = 5      # flag settings to report in explain
MINSUP   = 5      # a flag setting must occur in at least this many builds
LIFT     = 1.2    # ...and be this much more common among good builds

def _load(train_csv: str):
  "Table of TRAIN_CSV, with columns and goals from its header."
  return ezr.Tbl(ezr.csv(train_csv))

def _coerce(v):
  "Strings become atoms; other values pass through."
  return ezr.atom(v) if isinstance(v, str) else v

def _full(tbl, row) -> list:
  """Lift ROW into a full-width row. Accepts a {column name: value} dict,
  a full-width list, or a list with goal columns removed (header order).
  Any cell not given is '?'."""
  names = tbl.names
  out   = ["?"] * len(names)
  if isinstance(row, dict):
    for at, n in enumerate(names):
      if n in row: out[at] = _coerce(row[n])
    return out
  row = list(row)
  if len(row) == len(names):
    return [_coerce(v) for v in row]
  keep = [i for i, n in enumerate(names) if n[-1] not in "+-!"]
  for i, v in zip(keep, row): out[i] = _coerce(v)
  return out

def _ydist(tbl, row) -> float | None:
  "Distance of ROW's numeric goals from their best; None if no goal is known."
  sq, n = 0.0, 0
  for c in tbl.y:
    if type(c) is not ezr.Num: continue
    v = row[c.at]
    if v == "?": continue
    sq += (ezr.norm(c, v) - c.goal) ** 2
    n  += 1
  return (sq / n) ** .5 if n else None

def _cut(tbl):
  "Sorted known goal distances of TBL's rows, and the distance cutoff for good."
  ys  = sorted(d for d in (_ydist(tbl, r) for r in tbl.rows) if d is not None)
  thr = ys[int(QUANTILE * (len(ys) - 1))] if ys else float("inf")
  return ys, thr

def _goal_text(c) -> str:
  "Goal name without its suffix, and its direction."
  return f"{c.txt[:-1]} ({'maximise' if c.goal else 'minimise'})"

def _paths(tbl, node, conds: tuple = ()):
  "Yield (leaf, conditions) for every leaf under NODE, conditions in words."
  if not node.kids:
    yield node, conds
    return
  name    = tbl.names[node.at]
  is_sym  = type(tbl.cols[node.at]) is ezr.Sym
  yes     = f"{name} {'=' if is_sym else '<='} {ezr.say(node.v)}"
  no      = f"{name} {'!=' if is_sym else '>'} {ezr.say(node.v)}"
  yield from _paths(tbl, node.kids[0], conds + (yes,))
  yield from _paths(tbl, node.kids[1], conds + (no,))

def explain(train_csv) -> str:
  "Plain reasons for what makes builds good; each can be checked by counting."
  tbl     = _load(train_csv)
  ys, thr = _cut(tbl)
  if not ys:
    return "No training build has known goal values, so nothing can be explained."
  n    = len(tbl.rows)
  good = [d is not None and d <= thr
          for d in (_ydist(tbl, r) for r in tbl.rows)]
  base = sum(good) / n
  goals = ", ".join(_goal_text(c) for c in tbl.y)
  out  = [f"A build is good when its goals ({goals}) are in the best "
          f"{round(100 * QUANTILE)}% of the {n} training builds. Distance 0 is "
          f"ideal; good means distance at or below {ezr.say(thr)}, and the mean "
          f"over all builds is {ezr.say(sum(ys) / len(ys))}."]

  complete = [r for r in tbl.rows if all(r[c.at] != "?" for c in tbl.y)]
  conds: tuple = ()
  leaf = None
  if complete:
    ls   = list(_paths(tbl, ezr.tree(tbl, complete)))
    pool = [p for p in ls if len(p[0].rows) >= ezr.the.Leaf] or ls
    leaf, conds = min(pool, key=lambda p: p[0].mu)
  if conds:
    out.append(f"The clearest group is where {' and '.join(conds)}: "
               f"{len(leaf.rows)} of {len(complete)} builds, with mean "
               f"distance {ezr.say(leaf.mu)}.")
  else:
    out.append("No single input setting splits the builds into a clearly "
               "better group, so no rule is claimed.")

  hits = []
  for col in tbl.x:
    for v in sorted({r[col.at] for r in tbl.rows if r[col.at] != "?"}, key=str):
      seen = [g for r, g in zip(tbl.rows, good) if r[col.at] == v]
      if len(seen) < MINSUP or base == 0: continue
      rate = sum(seen) / len(seen)
      if rate / base >= LIFT:
        hits.append((rate / base, len(seen), col.txt, v, rate))
  hits.sort(key=lambda h: (-h[0], -h[1]))
  if hits:
    out.append("Input settings more common in good builds (checkable by counting):")
    for _, k, name, v, rate in hits[:TOP]:
      out.append(f"  {name} = {v}: {rate:.0%} of the {k} builds with it are "
                 f"good, versus {base:.0%} overall.")
  out.append("These are associations in the training data, not proof that "
             "a setting causes the goals.")
  return "\n".join(out)

def why(train_csv, row) -> str:
  "One sentence: is this build good or bad, and why, judged by its goals."
  tbl     = _load(train_csv)
  ys, thr = _cut(tbl)
  if not ys:
    return ("This build cannot be rated: the training file has no known "
            "goal values.")
  full  = _full(tbl, row)
  known = [(c, full[c.at]) for c in tbl.y if full[c.at] != "?"]
  if not known:
    return "This build cannot be rated: it gives no goal values."
  d = _ydist(tbl, full)
  if d is None:
    return "This build cannot be rated: its goal values are not numeric."
  verdict = "good" if d <= thr else "bad"
  beat    = sum(1 for y in ys if y > d) / len(ys)
  vals    = ", ".join(f"{c.txt[:-1]}={ezr.say(v)}" for c, v in known)
  return (f"This build is {verdict}: its goals ({vals}) sit closer to the ideal "
          f"than {beat:.0%} of the {len(ys)} training builds (distance "
          f"{ezr.say(d)}, good means at or below {ezr.say(thr)}).")
```

```python
# step7.py
"Step 7: choose. Find the best build while benchmarking at most BUDGET builds."
import math
import random
import ezr

START = 4      # builds labelled at random before any guided choice

def _val(v):
  "Strings become atoms; other values pass through."
  return ezr.atom(v) if isinstance(v, str) else v

def _name(s: str) -> str:
  "A goal name without its +, - or ! suffix."
  return s[:-1] if s[-1:] in "+-!" else s

def _goals(tbl, goal) -> list:
  "Numeric goal columns to optimise: all of them, or the one named GOAL."
  ys = list(tbl.y)
  if goal is not None:
    want  = goal.lower()
    tests = (lambda c: c.txt == goal,
             lambda c: c.txt.lower() == want,
             lambda c: _name(c.txt).lower() == _name(want),
             lambda c: _name(c.txt).lower().startswith(_name(want)))
    hits = []
    for test in tests:
      hits = [c for c in ys if test(c)]
      if hits: break
    if len(hits) != 1:
      raise ValueError(f"choose: goal {goal!r} must name exactly one goal")
    ys = hits
  ys = [c for c in ys if type(c) is ezr.Num]
  if not ys: raise ValueError("choose: no numeric goal to optimise")
  return ys

def _scores(goals, known: dict) -> dict:
  "Each labelled build's distance from the ideal, over the chosen goals only."
  norms = {}
  for c in goals:
    num = ezr.Num(c.txt, c.at)
    for vals in known.values():
      v = vals.get(c.txt)
      if isinstance(v, (int, float)): ezr.add(num, v)
    norms[c.txt] = num
  out = {}
  for i, vals in known.items():
    sq, n = 0.0, 0
    for c in goals:
      v = vals.get(c.txt)
      if not isinstance(v, (int, float)): continue
      sq += (ezr.norm(norms[c.txt], v) - c.goal) ** 2
      n  += 1
    out[i] = (sq / n) ** .5 if n else math.inf
  return out

def choose(blind_csv, label, budget=30, goal=None) -> int:
  "Index of the best build found, after labelling at most BUDGET builds."
  tbl  = ezr.Tbl(ezr.csv(blind_csv))
  rows = tbl.rows
  if not rows: raise ValueError("choose: no builds to choose from")
  if budget < 1: raise ValueError("choose: budget must be at least 1")
  goals  = _goals(tbl, goal)
  budget = min(budget, len(rows))
  known: dict[int, dict] = {}
  near:  dict[int, list[float]] = {}
  spent  = 0

  def ask(i: int) -> None:
    nonlocal spent
    if i in known: return
    if spent >= budget: raise RuntimeError("choose: budget exhausted")
    spent += 1
    known[i] = {k: _val(v) for k, v in label(i).items()}

  def gaps(b: int) -> list[float]:
    "Input distance from labelled build B to every build (cached)."
    if b not in near:
      near[b] = [ezr.xdist(tbl, z, rows[b]) for z in rows]
    return near[b]

  def lure(z: int, best: list, rest: list) -> float:
    "Closer to the best labelled builds than to the rest: lower is better."
    nb = min(gaps(b)[z] for b in best)
    return nb - min(gaps(b)[z] for b in rest) if rest else nb

  order = random.Random(ezr.the.Seed).sample(range(len(rows)), len(rows))
  for i in order[:min(START, budget)]: ask(i)

  while spent < budget:
    scores = _scores(goals, known)
    ranked = sorted(known, key=lambda i: (scores[i], i))
    k      = max(1, int(math.sqrt(len(ranked))))
    best, rest = ranked[:k], ranked[k:]
    pick = min((z for z in range(len(rows)) if z not in known),
               key=lambda z: (lure(z, best, rest), z))
    ask(pick)

  scores = _scores(goals, known)
  return min(known, key=lambda i: (scores[i], i))
```

```python
# step8.py
"Step 8: fix. The fewest input changes that make a bad build good, leaving frozen columns alone."
import ezr

QUANTILE = 0.25   # a build is "good" if its goals are in the best 25% of train
K        = 20     # nearest training rows consulted to predict a build's goals

_CACHE: dict = {}

def _coerce(v):
  "Strings become atoms; other values pass through."
  return ezr.atom(v) if isinstance(v, str) else v

def _full(tbl, row) -> list:
  """Lift ROW into a full-width row. Accepts a {column name: value} dict,
  a full-width list, or a list with goal columns removed (header order).
  Any cell not given is '?'."""

  out   = ["?"] * len(names)
  if isinstance(row, dict):
    for at, n in enumerate(names):
      if n in row: out[at] = _coerce(row[n])
    return out
  row = list(row)
  if len(row) == len(names):
    return [_coerce(v) for v in row]
  keep = [i for i, n in enumerate(names) if n[-1] not in "+-!"]
  for i, v in zip(keep, row): out[i] = _coerce(v)
  return out

def _ydist(tbl, row) -> float | None:
  "Distance of ROW's numeric goals from their best; None if no goal is known."
  sq, n = 0.0, 0
  for c in tbl.y:
    if type(c) is not ezr.Num: continue
    v = row[c.at]
    if v == "?": continue
    sq += (ezr.norm(c, v) - c.goal) ** 2
    n  += 1
  return (sq / n) ** .5 if n else None

def _load(train_csv: str):
  "Train table and the distance cutoff for good builds; cached per file."
  if train_csv not in _CACHE:
    tbl = ezr.Tbl(ezr.csv(train_csv))
    ys  = sorted(d for d in (_ydist(tbl, r) for r in tbl.rows)
                 if d is not None)
    thr = ys[int(QUANTILE * (len(ys) - 1))] if ys else float("inf")
    _CACHE[train_csv] = (tbl, thr)
  return _CACHE[train_csv]

def _predict(tbl, q: list) -> float | None:
  """Predicted goal distance of input vector Q: the median goal value of
  the K training rows nearest to Q over the input columns."""
  near  = sorted(tbl.rows, key=lambda r: ezr.xdist(tbl, q, r))[:K]
  guess = ["?"] * len(q)
  for c in tbl.y:
    if type(c) is not ezr.Num: continue
    xs = sorted(r[c.at] for r in near if r[c.at] != "?")
    if xs: guess[c.at] = xs[len(xs) // 2]
  return _ydist(tbl, guess)

def _inputs(tbl, v: list) -> dict:
  "Input columns of full-width row V, by name."
  return {c.txt: v[c.at] for c in tbl.x}

def fix(train_csv, row, frozen=()) -> dict:
  """Inputs of bad ROW with as few values changed as possible so that it
  becomes good. Columns named in FROZEN never change. Returns
  {input column name: value}."""
  tbl, thr = _load(train_csv)
  if isinstance(frozen, str): frozen = (frozen,)
  frozen_set = set(frozen)
  free = [c for c in tbl.x if c.txt not in frozen_set]
  q    = _full(tbl, row)

  def good(v: list) -> bool:
    d = _predict(tbl, v)
    return d is not None and d <= thr

  if good(q):                          # already good: change nothing
    return _inputs(tbl, q)

  # Candidates: each good training build, with its differing free inputs
  # copied in. Fewest copied values come first.
  cands = []
  for g in tbl.rows:
    gd = _ydist(tbl, g)
    if gd is None or gd > thr: continue
    v, changed = list(q), 0
    for c in free:
      gv = g[c.at]
      if gv != "?" and gv != q[c.at]:
        v[c.at] = gv
        changed += 1
    cands.append((changed, gd, v))
  cands.sort(key=lambda t: (t[0], t[1]))

  best = None
  for _, _, v in cands:
    if good(v):
      best = v
      break

  if best is None:                     # nothing predicted good: do the best we can
    options = [q] + [v for _, _, v in cands]
    best = min(options, key=lambda v: (_predict(tbl, v)
                                       if _predict(tbl, v) is not None
                                       else float("inf")))
    return _inputs(tbl, best)

  # Greedy pruning: undo each change that is not needed for goodness.
  for c in free:
    if best[c.at] != q[c.at]:
      trial = list(best)
      trial[c.at] = q[c.at]
      if good(trial): best = trial

  return _inputs(tbl, best)
```

```python
"Step 9: odd. Flag benchmark results that look like glitches, and flag flag sets unlike anything in train."
import math
import ezr

ZCAP   = 5.0    # goal z-scores are capped here, then scaled to 0..1
TYPO   = 1.0    # penalty per cell that should be a finite number but is not
UNSEEN = 1.0    # weight of the fraction of input symbols never seen in train

_CACHE: dict = {}

def _coerce(v):
  "Strings become atoms; other values pass through."
  return ezr.atom(v) if isinstance(v, str) else v

def _full(tbl, row) -> list:
  """Lift ROW into a full-width row. Accepts a {column name: value} dict,
  a full-width list, or a list with goal columns removed (header order).
  Any cell not given is '?'."""
  names = tbl.names
  out   = ["?"] * len(names)
  if isinstance(row, dict):
    for at, n in enumerate(names):
      if n in row: out[at] = _coerce(row[n])
    return out
  row = list(row)
  if len(row) == len(names):
    return [_coerce(v) for v in row]
  keep = [i for i, n in enumerate(names) if n[-1] not in "+-!"]
  for i, v in zip(keep, row): out[i] = _coerce(v)
  return out

def _load(train_csv: str):
  "Train table, and the symbol values seen per input column; cached per file."
  if train_csv not in _CACHE:
    tbl  = ezr.Tbl(ezr.csv(train_csv))
    seen = {c.at: set(c.has) for c in tbl.x if type(c) is ezr.Sym}
    _CACHE[train_csv] = (tbl, seen)
  return _CACHE[train_csv]

def _clean(tbl, q: list) -> tuple[list, int]:
  """Replace non-numeric or non-finite cells in numeric columns with '?',
  and count them. Such cells are typos or glitches."""
  typos = 0
  for c in list(tbl.x) + list(tbl.y):
    if type(c) is not ezr.Num: continue
    v = q[c.at]
    if v == "?": continue
    if not (isinstance(v, (int, float)) and math.isfinite(v)):
      q[c.at] = "?"
      typos  += 1
  return q, typos

def _score(tbl, seen: dict, row) -> float:
  "Suspicion of one ROW: far from every train row, odd goals, or typos."
  q, typos = _clean(tbl, _full(tbl, row))
  if tbl.x:
    nn = min((ezr.xdist(tbl, q, t) for t in tbl.rows), default=1.0)
  else:
    nn = 0.0
  unseen = sum(1 for c in tbl.x
               if type(c) is ezr.Sym and q[c.at] != "?"
               and q[c.at] not in seen[c.at])
  unseen = unseen / max(1, len(tbl.x))
  zs = []
  for c in tbl.y:
    v = q[c.at]
    if type(c) is ezr.Num and v != "?" and c.n > 1:
      z = abs(v - c.mu) / (c.sd + 1e-32)
      zs.append(min(ZCAP, z) / ZCAP)
  goal = max(zs, default=0.0)
  return float(nn + goal + UNSEEN * unseen + TYPO * typos)

def odd(train_csv, rows) -> list[float]:
  """One suspicion score per row in ROWS, in order. Higher means more likely
  a glitch, a typo, or unlike anything in TRAIN_CSV."""
  tbl, seen = _load(train_csv)
  return [_score(tbl, seen, r) for r in rows]
```
