### step1.py
```python
"""step1.py - client questions 1 and 2.

1. families(train_csv): cluster the flag combinations (input columns) into a
   small number of families. The number of families is chosen from 2..6 by
   silhouette score.
2. best(train_csv): rows that are not dominated on all goals together
   (Pareto-optimal, with goals taken as maximise for "+" and minimise for "-"),
   capped at 5% of rows.

The earlier steps' modules were not available in this message, so the small
helpers needed here (MOOT csv loading, k-means, silhouette, non-dominated
sorting) are implemented below with the standard library only.
"""
import csv
import math
import random

SEED = 0
MIN_FAMILIES = 2
MAX_FAMILIES = 6
BEST_FRACTION = 0.05


# --------------------------------------------------------------------------
# MOOT csv loading
# --------------------------------------------------------------------------
def _load(csv_path):
    """Return (input_cols, goal_cols, X, Y).

    X[i] is the list of input values of row i.
    Y[i] is the list of goal values of row i, with every goal turned into a
    maximisation goal (minimised goals are negated).
    """
    with open(csv_path, newline="") as f:
        reader = csv.DictReader(f)
        cols = reader.fieldnames or []
        input_cols = []
        goal_cols = []  # (name, sign): sign +1 maximise, -1 minimise
        for c in cols:
            if c.endswith("X"):
                continue
            if c.endswith("+"):
                goal_cols.append((c, +1))
            elif c.endswith("-"):
                goal_cols.append((c, -1))
            else:
                input_cols.append(c)
        X, Y = [], []
        for row in reader:
            X.append([float(row[c]) for c in input_cols])
            Y.append([sign * float(row[c]) for c, sign in goal_cols])
    return input_cols, goal_cols, X, Y


# --------------------------------------------------------------------------
# distances, k-means, silhouette
# --------------------------------------------------------------------------
def _sqdist(a, b):
    return sum((x - y) * (x - y) for x, y in zip(a, b))


def _kmeans(X, k, rng, iters=100):
    n, d = len(X), len(X[0])

    # k-means++ seeding
    centers = [list(X[rng.randrange(n)])]
    dist2 = [_sqdist(x, centers[0]) for x in X]
    while len(centers) < k:
        total = sum(dist2)
        if total <= 0:
            c = X[rng.randrange(n)]
        else:
            r = rng.uniform(0, total)
            acc = 0.0
            idx = n - 1
            for i, w in enumerate(dist2):
                acc += w
                if acc >= r:
                    idx = i
                    break
            c = X[idx]
        centers.append(list(c))
        dist2 = [min(dist2[i], _sqdist(X[i], c)) for i in range(n)]

    # Lloyd iterations
    labels = None
    for _ in range(iters):
        new = [min(range(k), key=lambda j: _sqdist(x, centers[j])) for x in X]
        if new == labels:
            break
        labels = new
        for j in range(k):
            members = [X[i] for i in range(n) if labels[i] == j]
            if members:
                centers[j] = [sum(m[t] for m in members) / len(members)
                              for t in range(d)]
            else:
                # reseed an empty cluster at the point farthest from its centre
                far = max(range(n),
                          key=lambda i: _sqdist(X[i], centers[labels[i]]))
                centers[j] = list(X[far])

    inertia = sum(_sqdist(X[i], centers[labels[i]]) for i in range(n))
    return labels, inertia


def _best_kmeans(X, k, rng, restarts=4):
    best = None
    for _ in range(restarts):
        labels, inertia = _kmeans(X, k, rng)
        if best is None or inertia < best[1]:
            best = (labels, inertia)
    return best[0]


def _silhouette(D, labels):
    n = len(labels)
    clusters = sorted(set(labels))
    if len(clusters) < 2:
        return -1.0
    members = {c: [i for i in range(n) if labels[i] == c] for c in clusters}
    total = 0.0
    for i in range(n):
        own = members[labels[i]]
        if len(own) == 1:
            continue  # silhouette of a singleton is 0
        a = sum(D[i][j] for j in own if j != i) / (len(own) - 1)
        b = min(sum(D[i][j] for j in members[c]) / len(members[c])
                for c in clusters if c != labels[i])
        m = max(a, b)
        if m > 0:
            total += (b - a) / m
    return total / n


# --------------------------------------------------------------------------
# dominance, fronts, crowding
# --------------------------------------------------------------------------
def _dominates(a, b):
    """True if a is at least as good as b on every goal (all maximised)
    and strictly better on at least one."""
    better = False
    for x, y in zip(a, b):
        if x < y:
            return False
        if x > y:
            better = True
    return better


def _crowding(Y, front):
    dist = {i: 0.0 for i in front}
    if len(front) <= 2:
        return {i: math.inf for i in front}
    for t in range(len(Y[0])):
        order = sorted(front, key=lambda i: Y[i][t])
        lo, hi = Y[order[0]][t], Y[order[-1]][t]
        dist[order[0]] = math.inf
        dist[order[-1]] = math.inf
        span = hi - lo
        if span == 0:
            continue
        for p in range(1, len(order) - 1):
            dist[order[p]] += (Y[order[p + 1]][t] - Y[order[p - 1]][t]) / span
    return dist


# --------------------------------------------------------------------------
# step 1 public functions
# --------------------------------------------------------------------------
def families(train_csv):
    """One family (cluster) label per data row of train_csv, in file order."""
    _, _, X, _ = _load(train_csv)
    n = len(X)
    if n < 3:
        return [0] * n

    D = [[0.0] * n for _ in range(n)]
    for i in range(n):
        for j in range(i + 1, n):
            D[i][j] = D[j][i] = math.sqrt(_sqdist(X[i], X[j]))

    rng = random.Random(SEED)
    best_labels, best_score = None, -math.inf
    for k in range(MIN_FAMILIES, min(MAX_FAMILIES, n - 1) + 1):
        labels = _best_kmeans(X, k, rng)
        score = _silhouette(D, labels)
        if score > best_score:
            best_labels, best_score = labels, score

    if best_labels is None:
        best_labels = [0] * n
    return [int(l) for l in best_labels]


def best(train_csv):
    """Row indices of the unusually good rows (not dominated on all goals
    together), at most BEST_FRACTION of the rows. Returned in ascending order."""
    _, _, _, Y = _load(train_csv)
    n = len(Y)
    cap = int(BEST_FRACTION * n)
    if n == 0 or cap == 0:
        return []

    chosen = []
    remaining = list(range(n))
    while remaining and len(chosen) < cap:
        # current non-dominated front among the remaining rows
        front = [i for i in remaining
                 if not any(_dominates(Y[j], Y[i]) for j in remaining if j != i)]
        if not front:
            front = list(remaining)
        fs = set(front)
        remaining = [i for i in remaining if i not in fs]

        room = cap - len(chosen)
        if len(front) > room:
            # keep the most spread-out members of the partial front
            dist = _crowding(Y, front)
            front = sorted(front, key=lambda i: (-dist[i], i))[:room]
        chosen.extend(front)

    return sorted(chosen)
```

### step2.py
```python
"""step2.py - client question 3: guess the goals of a flag set.

guess(train_csv, row) estimates every goal of an unseen flag set from the
training rows whose flag sets are most similar to it (k nearest neighbours
in the standardised flag space). The estimate is the median of the
neighbours' goal values. low..high is the 10th..90th percentile range of
those neighbours, so the range widens when similar flag sets disagree.

Goal keys are the goal column names as they appear in the csv, suffix
included (e.g. "Performance-", "Cpu-"). Values are in original units.

Loading and distance helpers come from step1.
"""
import functools
import math
import statistics

from step1 import _load, _sqdist

K_NEIGHBOURS = 10
LOW_Q = 0.10
HIGH_Q = 0.90


@functools.lru_cache(maxsize=None)
def _model(train_csv):
    """Load the training data once per path and precompute what guess needs."""
    input_cols, goal_cols, X, Y = _load(train_csv)
    n = len(X)
    d = len(input_cols)

    means = [0.0] * d
    stds = [1.0] * d
    if n:
        for t in range(d):
            means[t] = sum(x[t] for x in X) / n
            var = sum((x[t] - means[t]) ** 2 for x in X) / n
            s = math.sqrt(var)
            stds[t] = s if s > 0 else 1.0

    Z = [[(x[t] - means[t]) / stds[t] for t in range(d)] for x in X]

    # _load negates minimised goals; undo that to get original units.
    goal_vals = [
        [sign * Y[i][g] for g, (_, sign) in enumerate(goal_cols)]
        for i in range(n)
    ]
    return input_cols, goal_cols, means, stds, Z, goal_vals


def _quantile(sorted_vals, q):
    """Linear-interpolated quantile of an already sorted, non-empty list."""
    if len(sorted_vals) == 1:
        return sorted_vals[0]
    pos = q * (len(sorted_vals) - 1)
    lo = int(math.floor(pos))
    hi = min(lo + 1, len(sorted_vals) - 1)
    frac = pos - lo
    return sorted_vals[lo] + (sorted_vals[hi] - sorted_vals[lo]) * frac


def guess(train_csv, row):
    """Estimate every goal of a flag set.

    `row` maps input column names to values (goal columns removed).
    Returns {goal: (estimate, low, high)} for every goal column.
    """
    input_cols, goal_cols, means, stds, Z, goal_vals = _model(train_csv)
    n = len(Z)
    if n == 0:
        raise ValueError("training file has no data rows")

    missing = [c for c in input_cols if c not in row]
    if missing:
        raise KeyError(f"row is missing input columns: {missing}")

    z = [(float(row[c]) - means[t]) / stds[t] for t, c in enumerate(input_cols)]

    # k nearest training rows; ties broken by row index for determinism
    order = sorted(range(n), key=lambda i: (_sqdist(Z[i], z), i))
    nbrs = order[:min(K_NEIGHBOURS, n)]

    result = {}
    for g, (name, _) in enumerate(goal_cols):
        vals = sorted(goal_vals[i][g] for i in nbrs)
        est = statistics.median(vals)
        low = min(_quantile(vals, LOW_Q), est)
        high = max(_quantile(vals, HIGH_Q), est)
        result[name] = (float(est), float(low), float(high))
    return result
```

### step3.py
```python
"""step3.py - client question 4: is a difference real or just noise?

same(xs, ys) decides whether two lists of measurements (e.g. run times)
differ by more than run-to-run noise.

It runs a two-sided permutation test on the difference of means. If the
two groups came from the same distribution, shuffling the group labels
would often produce a gap at least as large as the observed one. When
that p-value is at least ALPHA, the gap is called noise. The test makes no
distributional assumption, so it suits skewed run-time data.

SEED is reused from step1 so results are reproducible.
"""
import random

from step1 import SEED

ALPHA = 0.05
PERMUTATIONS = 2000


def same(xs, ys):
    """True if the gap between xs and ys is noise, False if it is real."""
    xs = [float(x) for x in xs]
    ys = [float(y) for y in ys]
    if not xs or not ys:
        return True  # no evidence of any difference

    nx, ny = len(xs), len(ys)
    total = sum(xs) + sum(ys)
    observed = abs(sum(xs) / nx - sum(ys) / ny)
    if observed == 0:
        return True

    rng = random.Random(SEED)
    pool = xs + ys
    hits = 0
    for _ in range(PERMUTATIONS):
        rng.shuffle(pool)
        s = sum(pool[:nx])
        gap = abs(s / nx - (total - s) / ny)
        if gap >= observed:
            hits += 1

    p_value = (hits + 1) / (PERMUTATIONS + 1)
    return p_value >= ALPHA
```

### step4.py
```python
"""step4.py - client question 5: flow of nightly benchmark results.

stream(train_csv, rows) keeps a row (True) when no reference point is better
on every goal. The reference points are the training rows plus the rows that
came before it in this stream. Only train_csv and earlier rows are used.

drift(train_csv, rows) warns at the first row where the recent window of rows
has stopped looking like train_csv in flag space. A new row is "out of place"
if its nearest training flag set (standardised as in step2) is farther away
than the OUT_QUANTILE of the training rows' own leave-one-out nearest-neighbour
distances. The warning fires when at least OUT_FRACTION of the last WINDOW rows
are out of place. Under the training distribution that many out-of-place rows
in one window is very unlikely, so ordinary variation does not trigger it.

Helpers are reused from step1 (_load, _sqdist, _dominates) and step2
(_model, _quantile).
"""
import collections
import functools
import math

from step1 import _dominates, _load, _sqdist
from step2 import _model, _quantile

WINDOW = 20
OUT_QUANTILE = 0.99
OUT_FRACTION = 0.5


@functools.lru_cache(maxsize=None)
def _train_goals(train_csv):
    """Goal columns and training goal vectors, maximisation-signed."""
    _, goal_cols, _, Y = _load(train_csv)
    return goal_cols, [list(y) for y in Y]


def _goal_vector(row, goal_cols):
    """Maximisation-signed goal vector of one row."""
    return [sign * float(row[name]) for name, sign in goal_cols]


def stream(train_csv: str, rows) -> list[bool]:
    """Keep (True) each row that is not dominated on all goals by any
    training row or by any earlier row of this stream."""
    goal_cols, train_Y = _train_goals(train_csv)
    reference = list(train_Y)
    keep = []
    for row in rows:
        y = _goal_vector(row, goal_cols)
        dominated = any(_dominates(r, y) for r in reference)
        keep.append(not dominated)
        reference.append(y)
    return keep


@functools.lru_cache(maxsize=None)
def _threshold(train_csv):
    """OUT_QUANTILE of the leave-one-out nearest-neighbour squared distances
    among training flag sets, or None if there are fewer than two rows."""
    _, _, _, _, Z, _ = _model(train_csv)
    n = len(Z)
    if n < 2:
        return None
    nn = []
    for i in range(n):
        nn.append(min(_sqdist(Z[i], Z[j]) for j in range(n) if j != i))
    nn.sort()
    return _quantile(nn, OUT_QUANTILE)


def _nearest_sq(z, Z):
    return min(_sqdist(z, zi) for zi in Z)


def drift(train_csv: str, rows) -> int | None:
    """Index into rows of the first row at which the recent window stops
    looking like train_csv, or None if it never does."""
    input_cols, _, means, stds, Z, _ = _model(train_csv)
    thr = _threshold(train_csv)
    if thr is None or not Z:
        return None

    need = math.ceil(OUT_FRACTION * WINDOW)
    window = collections.deque(maxlen=WINDOW)
    count = 0
    for i, row in enumerate(rows):
        z = [(float(row[c]) - means[t]) / stds[t]
             for t, c in enumerate(input_cols)]
        out = _nearest_sq(z, Z) > thr
        if len(window) == WINDOW:
            count -= window[0]
        window.append(out)
        count += out
        if len(window) == WINDOW and count >= need:
            return i
    return None
```

### step5.py
```python
"""step5.py - client question 6: blame.

blame(train_csv) ranks the input columns (flags/passes) by how strongly they
are tied to bad rows, where a row is bad when it is poor on the goals.

- Each goal is turned into a percentile rank among the training rows (0 =
  worst, 1 = best), so goals with different units count equally. Goals come
  from step1._load, which already makes minimised goals maximisation-signed.
- A row's badness is 1 minus its mean percentile over all goals. The worst
  BAD_FRACTION of rows by badness are "bad"; the rest are "good".
- For each input column, the effect is the difference between its mean among
  bad rows and among good rows, divided by the pooled within-group standard
  deviation. This standardised mean difference makes 0/1 flags and numeric
  inputs comparable. A positive effect means the input is on, or higher, more
  often in bad rows.
- Columns are returned by effect, largest first. Ties keep file order.
"""
import math

from step1 import _load

BAD_FRACTION = 0.25


def _percentiles(values):
    """Percentile rank of each value in [0, 1] (0 = smallest). Ties share the
    mean rank of their group."""
    n = len(values)
    if n == 1:
        return [1.0]
    order = sorted(range(n), key=lambda i: values[i])
    pct = [0.0] * n
    p = 0
    while p < n:
        q = p
        while q + 1 < n and values[order[q + 1]] == values[order[p]]:
            q += 1
        mid = (p + q) / 2
        for k in range(p, q + 1):
            pct[order[k]] = mid / (n - 1)
        p = q + 1
    return pct


def _effect(bad_vals, good_vals):
    """Standardised mean difference, bad minus good."""
    nb, ng = len(bad_vals), len(good_vals)
    mb = sum(bad_vals) / nb
    mg = sum(good_vals) / ng
    ss = (sum((x - mb) ** 2 for x in bad_vals)
          + sum((x - mg) ** 2 for x in good_vals))
    sd = math.sqrt(ss / (nb + ng))
    if sd == 0:
        return 0.0
    return (mb - mg) / sd


def blame(train_csv) -> list[str]:
    """Input column names, most to blame for bad rows (bad on the goals) first."""
    input_cols, goal_cols, X, Y = _load(train_csv)
    n = len(X)
    d = len(input_cols)
    m = len(goal_cols)
    if n == 0 or d == 0 or m == 0:
        return list(input_cols)

    # badness: 1 minus the mean percentile over all goals (1 = worst)
    pct_cols = [_percentiles([Y[i][g] for i in range(n)]) for g in range(m)]
    badness = [1.0 - sum(col[i] for col in pct_cols) / m for i in range(n)]

    k = max(1, int(BAD_FRACTION * n))
    order = sorted(range(n), key=lambda i: (-badness[i], i))
    bad_idx = sorted(order[:k])
    bad_set = set(bad_idx)
    good_idx = [i for i in range(n) if i not in bad_set]
    if not good_idx:
        return list(input_cols)

    effects = [
        _effect([X[i][t] for i in bad_idx], [X[i][t] for i in good_idx])
        for t in range(d)
    ]
    ranked = sorted(range(d), key=lambda t: (-effects[t], t))
    return [input_cols[t] for t in ranked]
```
