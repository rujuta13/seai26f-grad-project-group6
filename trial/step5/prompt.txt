Use only the Python 3.13 standard library. No third-party packages.

Data conventions:
- Data files are csv files (MOOT format). Column names ending in "+" are goals to maximise,
  "-" goals to minimise; names ending in "X" are ignored; every other column is an input.
- A *row* is a dict {column name: value}; numbers are floats, other cells are strings.
- Row indices count data rows of a csv from 0 (header excluded).
- `train_csv` / `test_csv` are paths to such files, about 800 / 200 rows.

### step1.py
```python
"""Step 1: flag-family clustering and best-row selection.

Standard library only. The earlier-step modules were not provided, so this
file is self-contained and does not import from them.
"""

import csv
import random


K_FAMILIES = 5           # number of flag families reported
SEED = 1                 # fixed seed so results are reproducible
KMEANS_ITERS = 100


def _load(train_csv):
    """Return (header, rows). Numeric cells become floats; others stay strings."""
    with open(train_csv, newline="") as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = []
        for cells in reader:
            if not cells or all(c.strip() == "" for c in cells):
                continue
            row = {}
            for name, cell in zip(header, cells):
                try:
                    row[name] = float(cell)
                except ValueError:
                    row[name] = cell
            rows.append(row)
    return header, rows


def _columns(header):
    """Split columns into (inputs, goals). Goals are (name, sign): +1 max, -1 min."""
    inputs, goals = [], []
    for name in header:
        if name.endswith("X"):
            continue
        if name.endswith("+"):
            goals.append((name, 1.0))
        elif name.endswith("-"):
            goals.append((name, -1.0))
        else:
            inputs.append(name)
    return inputs, goals


def _input_matrix(rows, inputs):
    """One numeric vector per row. Non-numeric cells are coded by first appearance."""
    codes = {}
    matrix = []
    for row in rows:
        vec = []
        for c in inputs:
            v = row[c]
            if isinstance(v, float):
                vec.append(v)
            else:
                table = codes.setdefault(c, {})
                vec.append(float(table.setdefault(v, len(table))))
        matrix.append(vec)
    return matrix


def _sqdist(a, b):
    return sum((x - y) * (x - y) for x, y in zip(a, b))


def _kmeans(points, k, rng, iters=KMEANS_ITERS):
    """Lloyd's algorithm with k-means++ seeding. Returns one label per point."""
    n = len(points)
    k = min(k, n)
    dim = len(points[0])

    # k-means++ seeding
    centers = [list(points[rng.randrange(n)])]
    while len(centers) < k:
        d = [min(_sqdist(p, c) for c in centers) for p in points]
        total = sum(d)
        if total == 0:
            centers.append(list(points[rng.randrange(n)]))
            continue
        r = rng.uniform(0, total)
        acc = 0.0
        pick = n - 1
        for i, di in enumerate(d):
            acc += di
            if acc >= r:
                pick = i
                break
        centers.append(list(points[pick]))

    labels = [-1] * n
    for _ in range(iters):
        changed = False
        for i, p in enumerate(points):
            best_c = min(range(k), key=lambda c: _sqdist(p, centers[c]))
            if best_c != labels[i]:
                labels[i] = best_c
                changed = True
        if not changed:
            break

        sums = [[0.0] * dim for _ in range(k)]
        counts = [0] * k
        for p, lab in zip(points, labels):
            counts[lab] += 1
            s = sums[lab]
            for j, x in enumerate(p):
                s[j] += x
        for c in range(k):
            if counts[c]:
                centers[c] = [v / counts[c] for v in sums[c]]
            else:
                # Reseed an empty cluster from a random point.
                centers[c] = list(points[rng.randrange(n)])
    return labels


def families(train_csv) -> list[int]:
    """One family (cluster) label per data row of train_csv, in file order."""
    header, rows = _load(train_csv)
    if not rows:
        return []
    inputs, _ = _columns(header)
    points = _input_matrix(rows, inputs)
    if not inputs:
        return [0] * len(rows)
    rng = random.Random(SEED)
    return _kmeans(points, K_FAMILIES, rng)


def _normalised_goals(rows, goals):
    """Per row, a vector of goal values in [0, 1] where higher is always better."""
    cols = []
    for name, sign in goals:
        vals = [sign * float(r[name]) for r in rows]
        lo, hi = min(vals), max(vals)
        span = hi - lo
        cols.append([(v - lo) / span if span > 0 else 0.0 for v in vals])
    return [list(t) for t in zip(*cols)]


def _dominates(a, b):
    """True if a is at least as good as b on every goal and strictly better on one."""
    return all(x >= y for x, y in zip(a, b)) and any(x > y for x, y in zip(a, b))


def _pareto_layers(points):
    """Non-dominated sorting. Returns the layer index of each point (0 = Pareto front)."""
    n = len(points)
    layer_of = [0] * n
    remaining = list(range(n))
    layer = 0
    while remaining:
        front = []
        for i in remaining:
            pi = points[i]
            if not any(_dominates(points[j], pi) for j in remaining if j != i):
                front.append(i)
        for i in front:
            layer_of[i] = layer
        front_set = set(front)
        remaining = [i for i in remaining if i not in front_set]
        layer += 1
    return layer_of


def best(train_csv) -> list[int]:
    """Row indices of rows that are best on all goals together (Pareto front),
    ranked by mean normalised goal value and capped at 5% of rows."""
    header, rows = _load(train_csv)
    n = len(rows)
    cap = n // 20  # floor(5% of n), computed without float error
    _, goals = _columns(header)
    if n == 0 or cap == 0 or not goals:
        return []

    points = _normalised_goals(rows, goals)
    layer_of = _pareto_layers(points)
    score = [sum(p) / len(p) for p in points]

    front = [i for i in range(n) if layer_of[i] == 0]
    front.sort(key=lambda i: (-score[i], i))
    return front[:cap]
```

### step2.py
```python
"""Step 2: guess the goal values of an unseen flag set.

Standard library only. Loading, column splitting, input encoding and the
squared-distance helper are reused from step1.

Method: inverse-distance-weighted k-nearest-neighbour regression on the
input columns. Each interval is the estimate shifted by the 5th and 95th
percentiles of leave-one-out residuals measured on the training data, so
the band reflects the error the model actually makes on this data.
"""

import heapq
import math

from step1 import _load, _columns, _input_matrix, _sqdist


K_NEIGHBOURS = 7
CALIB_SAMPLE = 200      # training rows used for leave-one-out calibration
BAND_LO = 0.05          # lower residual quantile for the interval
BAND_HI = 0.95          # upper residual quantile for the interval

_cache = {}


def _quantile(sorted_vals, q):
    """Linear-interpolated quantile of a non-empty sorted list."""
    pos = q * (len(sorted_vals) - 1)
    lo = math.floor(pos)
    hi = min(lo + 1, len(sorted_vals) - 1)
    frac = pos - lo
    return sorted_vals[lo] * (1.0 - frac) + sorted_vals[hi] * frac


def _predict(query, points, targets, exclude=None):
    """Inverse-distance-weighted mean of the nearest neighbours' goal values.

    Returns None when there is no neighbour to average over."""
    nearest = heapq.nsmallest(
        K_NEIGHBOURS,
        ((_sqdist(query, p), i) for i, p in enumerate(points) if i != exclude),
    )
    if not nearest:
        return None
    m = len(targets[0])
    est = [0.0] * m
    wsum = 0.0
    for d, i in nearest:
        w = 1.0 / (1.0 + d)
        wsum += w
        for g in range(m):
            est[g] += w * targets[i][g]
    return [v / wsum for v in est]


def _calibrate(points, targets, n_goals):
    """Per goal, the (lo, hi) residual quantiles of leave-one-out predictions."""
    n = len(points)
    step = max(1, n // CALIB_SAMPLE)
    resid = [[] for _ in range(n_goals)]
    for i in range(0, n, step):
        pred = _predict(points[i], points, targets, exclude=i)
        if pred is None:
            continue
        for g in range(n_goals):
            resid[g].append(targets[i][g] - pred[g])
    bands = []
    for vals in resid:
        if not vals:
            bands.append((-math.inf, math.inf))
            continue
        vals.sort()
        bands.append((_quantile(vals, BAND_LO), _quantile(vals, BAND_HI)))
    return bands


def _model(train_csv):
    """Load and cache the training data, its encoding and its calibration."""
    if train_csv in _cache:
        return _cache[train_csv]
    header, rows = _load(train_csv)
    inputs, goals = _columns(header)
    points = _input_matrix(rows, inputs)
    targets = [[float(r[name]) for name, _ in goals] for r in rows]
    bands = _calibrate(points, targets, len(goals))
    model = {
        "rows": rows,
        "inputs": inputs,
        "goals": goals,
        "points": points,
        "targets": targets,
        "bands": bands,
    }
    _cache[train_csv] = model
    return model


def guess(train_csv, row) -> dict[str, tuple[float, float, float]]:
    """Estimate every goal for an unseen flag set.

    `row` holds the input columns only (goal columns removed). Returns
    {goal: (estimate, low, high)} where low..high is the calibrated range
    the true value should fall in.
    """
    model = _model(train_csv)
    goals = model["goals"]
    targets = model["targets"]

    if not targets:
        return {name: (0.0, -math.inf, math.inf) for name, _ in goals}

    # Encode the query together with the training rows so that categorical
    # codes match the ones used for the training points.
    probe = {c: row.get(c, 0.0) for c in model["inputs"]}
    qpoint = _input_matrix(model["rows"] + [probe], model["inputs"])[-1]

    pred = _predict(qpoint, model["points"], targets)

    out = {}
    for g, (name, _) in enumerate(goals):
        est = pred[g]
        rlo, rhi = model["bands"][g]
        out[name] = (est, min(est, est + rlo), max(est, est + rhi))
    return out
```

### step3.py
```python
"""Step 3: certify whether two sets of measurements really differ.

Standard library only. The seed is reused from step1 so results are
reproducible.

Method: a two-sided permutation test on the difference of means. It makes
no normality assumption, which suits run times. The p-value is the share
of random relabellings of the pooled data whose mean gap is at least as
large as the observed gap (with the usual +1 correction).
"""

import random

from step1 import SEED


ALPHA = 0.05          # significance level: p <= ALPHA means a real difference
PERMUTATIONS = 4000   # random relabellings used to estimate the p-value


def same(xs, ys) -> bool:
    """True if the gap between xs and ys is noise, False if it is a real difference."""
    xs = [float(x) for x in xs]
    ys = [float(y) for y in ys]
    nx, ny = len(xs), len(ys)
    if nx == 0 or ny == 0:
        # Nothing to compare, so no difference can be claimed.
        return True

    observed = abs(sum(xs) / nx - sum(ys) / ny)
    if observed == 0.0:
        return True

    pooled = xs + ys
    total = sum(pooled)
    rng = random.Random(SEED)
    tol = observed * (1.0 - 1e-9)  # guard against floating-point ties

    extreme = 0
    for _ in range(PERMUTATIONS):
        rng.shuffle(pooled)
        sa = sum(pooled[:nx])
        gap = abs(sa / nx - (total - sa) / ny)
        if gap >= tol:
            extreme += 1

    p_value = (extreme + 1) / (PERMUTATIONS + 1)
    return p_value > ALPHA
```

### step4.py
```python
"""Step 4: streaming keep/drift decisions for nightly benchmark results.

Standard library only. Reuses the Pareto dominance and encoding helpers from
step1, the cached training model from step2, and the permutation test from
step3.

stream: a new row is kept when no training row and no earlier row of the
stream is at least as good on every goal, with one strictly better (or
identical). Goals are normalised with the bounds of train_csv, so the verdict
depends only on the training data and the rows already seen.

drift: each incoming row is summarised by its Euclidean distance to the
nearest training row in input space. Those distances are compared, in
non-overlapping blocks of DRIFT_BLOCK rows, with the leave-one-out nearest
neighbour distances of the training data, using step3.same. The index
returned is the last row of the first block that differs.
"""

import math

from step1 import _dominates, _input_matrix, _sqdist
from step2 import _model
from step3 import same


DRIFT_BLOCK = 30      # rows per non-overlapping drift check
CALIB_SAMPLE = 200    # training rows used for the reference distances

_ref_cache = {}


def _goal_bounds(model):
    """Per goal, the (lo, hi) of the signed training values (higher is better)."""
    bounds = []
    for g, (_, sign) in enumerate(model["goals"]):
        vals = [sign * t[g] for t in model["targets"]]
        if not vals:
            bounds.append((0.0, 0.0))
        else:
            bounds.append((min(vals), max(vals)))
    return bounds


def _goal_vector(values, goals, bounds):
    """Goal values of one row, scaled by the training bounds (higher is better).

    A goal with zero training span keeps its raw offset from the training
    value, so a different result on that goal is still visible."""
    vec = []
    for (name, sign), (lo, hi) in zip(goals, bounds):
        span = hi - lo
        scale = span if span > 0 else 1.0
        vec.append((sign * float(values[name]) - lo) / scale)
    return vec


def stream(train_csv, rows) -> list[bool]:
    """Keep (True) a row if no training row and no earlier stream row is at
    least as good on every goal with a strict gain on one (or an exact tie).
    Each decision uses only train_csv and the rows before it."""
    model = _model(train_csv)
    goals = model["goals"]
    if not goals:
        return [False for _ in rows]

    bounds = _goal_bounds(model)
    seen = [_goal_vector(r, goals, bounds) for r in model["rows"]]

    decisions = []
    for row in rows:
        vec = _goal_vector(row, goals, bounds)
        keep = not any(p == vec or _dominates(p, vec) for p in seen)
        decisions.append(keep)
        seen.append(vec)
    return decisions


def _nearest_sq(query, points, exclude=None):
    """Squared distance from query to its nearest point, skipping index exclude."""
    best = math.inf
    for i, p in enumerate(points):
        if i == exclude:
            continue
        d = _sqdist(query, p)
        if d < best:
            best = d
    return best


def _reference(train_csv, model):
    """Leave-one-out nearest-neighbour distances of a sample of training rows."""
    if train_csv in _ref_cache:
        return _ref_cache[train_csv]
    points = model["points"]
    n = len(points)
    step = max(1, n // CALIB_SAMPLE)
    ref = []
    for i in range(0, n, step):
        d = _nearest_sq(points[i], points, exclude=i)
        if d != math.inf:
            ref.append(math.sqrt(d))
    _ref_cache[train_csv] = ref
    return ref


def _encode(model, row):
    """Encode one row with the training codes, as step2.guess does."""
    probe = {c: row.get(c, 0.0) for c in model["inputs"]}
    return _input_matrix(model["rows"] + [probe], model["inputs"])[-1]


def drift(train_csv, rows) -> int | None:
    """Index into rows where new results first stop looking like train_csv,
    judged on input-space nearest-neighbour distances in blocks of
    DRIFT_BLOCK rows. Returns None if no block differs."""
    model = _model(train_csv)
    points = model["points"]
    if not points:
        return None
    ref = _reference(train_csv, model)
    if not ref:
        return None

    block = []
    for i, row in enumerate(rows):
        q = _encode(model, row)
        d = _nearest_sq(q, points)
        block.append(math.sqrt(d))
        if len(block) == DRIFT_BLOCK:
            if not same(block, ref):
                return i
            block = []
    return None
```
