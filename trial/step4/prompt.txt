- Data files are csv files (MOOT format). Column names ending in "+" are goals to maximise,
  "-" goals to minimise; names ending in "X" are ignored; every other column is an input.

- Row indices count data rows of a csv from 0 (header excluded).
- `train_csv` / `test_csv` are paths to such files, about 800 / 200 rows.

import csv
import math

import numpy as np
from sklearn.cluster import KMeans


def _load(train_csv):
    with open(train_csv, newline="") as f:
        reader = csv.DictReader(f)
        cols = list(reader.fieldnames or [])
        rows = list(reader)
    return cols, rows


def _to_float(value):
    try:
        return float(value)
    except (TypeError, ValueError):
        return math.nan


def _split_columns(cols):
    goal_max = [c for c in cols if c.endswith("+")]
    goal_min = [c for c in cols if c.endswith("-")]
    inputs = [c for c in cols if not c.endswith(("+", "-", "X"))]
    return inputs, goal_max, goal_min


def families(train_csv) -> list[int]:
    """One family (cluster) label per data row of train_csv, in file order."""
    cols, rows = _load(train_csv)
    n = len(rows)
    if n == 0:
        return []

    inputs, _, _ = _split_columns(cols)
    if not inputs:
        return [0] * n

    X = np.array(
        [[_to_float(r[c]) for c in inputs] for r in rows], dtype=float
    )
    X = np.where(np.isnan(X), 0.0, X)

    n_unique = len(np.unique(X, axis=0))
    k = min(5, n_unique)
    if k <= 1:
        return [0] * n

    model = KMeans(n_clusters=k, n_init=10, random_state=0)
    labels = model.fit_predict(X)
    return [int(x) for x in labels]


def best(train_csv) -> list[int]:
    """Row indices of the unusually good rows (best on all goals together),
    at most 5% of rows."""
    cols, rows = _load(train_csv)
    n = len(rows)
    cap = int(math.floor(0.05 * n))
    if n == 0 or cap == 0:
        return []

    _, goal_max, goal_min = _split_columns(cols)
    goals = goal_min + goal_max
    if not goals:
        return []

    # Build a matrix where every goal is minimised.
    G = np.empty((n, len(goals)), dtype=float)
    for j, c in enumerate(goals):
        vals = np.array([_to_float(r[c]) for r in rows], dtype=float)
        vals = np.where(np.isnan(vals), np.inf, vals)  # missing = worst
        if c.endswith("+"):
            vals = -vals
        G[:, j] = vals

    # Pareto front (non-dominated rows), all goals minimised.
    dominated = np.zeros(n, dtype=bool)
    for i in range(n):
        weakly = np.all(G <= G[i], axis=1)
        strictly = np.any(G < G[i], axis=1)
        dominated[i] = np.any(weakly & strictly)
    front = np.flatnonzero(~dominated)

    if len(front) > cap:
        # Too many non-dominated rows: keep those with the best average rank
        # across goals.
        sub = G[front]
        rank_sum = np.zeros(len(front), dtype=float)
        for j in range(sub.shape[1]):
            order = np.argsort(sub[:, j], kind="stable")
            ranks = np.empty(len(front), dtype=float)
            ranks[order] = np.arange(len(front), dtype=float)
            rank_sum += ranks
        keep = np.argsort(rank_sum, kind="stable")[:cap]
        front = front[keep]

    return sorted(int(i) for i in front)
```

### step2.py

import csv
import math

import numpy as np
from sklearn.ensemble import RandomForestRegressor


def _to_float(value):
    try:
        return float(value)
    except (TypeError, ValueError):
        return math.nan


def _load(train_csv):
    with open(train_csv, newline="") as f:
        reader = csv.DictReader(f)
        cols = list(reader.fieldnames or [])
        rows = list(reader)
    return cols, rows


def guess(train_csv, row) -> dict[str, tuple[float, float, float]]:
    """Estimate every goal for a flag set `row` (goal columns removed).

    Returns {goal: (estimate, low, high)} where low..high is an approximate
    90% interval built from out-of-bag residuals of a random forest.
    """
    cols, rows = _load(train_csv)
    goals = [c for c in cols if c.endswith(("+", "-"))]
    inputs = [c for c in cols if not c.endswith(("+", "-", "X"))]

    result: dict[str, tuple[float, float, float]] = {}

    if not rows or not goals:
        for g in goals:
            result[g] = (0.0, -math.inf, math.inf)
        return result

    # Feature matrix for training rows; missing cells become 0.0 (as in step1).
    X = np.array(
        [[_to_float(r.get(c, math.nan)) for c in inputs] for r in rows],
        dtype=float,
    ).reshape(len(rows), len(inputs))
    X = np.where(np.isnan(X), 0.0, X)

    # Feature vector for the query row.
    x = np.array(
        [[_to_float(row.get(c, math.nan)) for c in inputs]], dtype=float
    ).reshape(1, len(inputs))
    x = np.where(np.isnan(x), 0.0, x)

    for g in goals:
        y = np.array([_to_float(r.get(g, math.nan)) for r in rows], dtype=float)
        finite = np.isfinite(y)
        Xg, yg = X[finite], y[finite]

        if len(yg) == 0:
            result[g] = (0.0, -math.inf, math.inf)
            continue

        if len(yg) < 10 or not inputs:
            # Too little data for a model: use the empirical distribution.
            est = float(np.median(yg))
            lo = float(np.quantile(yg, 0.05))
            hi = float(np.quantile(yg, 0.95))
            result[g] = (est, min(lo, est), max(hi, est))
            continue

        model = RandomForestRegressor(
            n_estimators=300,
            min_samples_leaf=2,
            bootstrap=True,
            oob_score=True,
            random_state=0,
        )
        model.fit(Xg, yg)

        est = float(model.predict(x)[0])

        oob = np.asarray(model.oob_prediction_, dtype=float)
        ok = np.isfinite(oob)
        if ok.sum() >= 10:
            resid = yg[ok] - oob[ok]
            q_lo, q_hi = np.quantile(resid, [0.05, 0.95])
        else:
            q_lo, q_hi = np.quantile(yg, [0.05, 0.95]) - np.median(yg)

        low = est + float(q_lo)
        high = est + float(q_hi)
        low = min(low, est)
        high = max(high, est)
        result[g] = (est, low, high)

    return result
```

### step3.py
```python
import math

import numpy as np


def _clean(values):
    """Convert to a 1-D float array, dropping missing and non-finite entries."""
    out = []
    for v in values:
        try:
            f = float(v)
        except (TypeError, ValueError):
            continue
        if math.isfinite(f):
            out.append(f)
    return np.asarray(out, dtype=float)


def _average_ranks(values):
    """Ranks 1..N with ties replaced by their average rank."""
    uniq, inverse, counts = np.unique(values, return_inverse=True, return_counts=True)
    # Rank range for each unique value is [start+1, start+count]; average is midpoint.
    starts = np.cumsum(counts) - counts
    avg = starts + (counts + 1) / 2.0
    return avg[inverse]


def same(xs, ys) -> bool:
    """True if xs and ys are NOT really different (the gap is noise), False otherwise.

    Uses a two-sided permutation test on the difference in mean rank (a
    Mann-Whitney-style test). It makes no distributional assumption, so it is
    robust to skewed run times and outliers. The test is deterministic (fixed
    seed). The significance level is 0.05.
    """
    a = _clean(xs)
    b = _clean(ys)
    n, m = len(a), len(b)
    if n == 0 or m == 0:
        # No evidence of any difference.
        return True

    pooled = np.concatenate([a, b])
    if np.all(pooled == pooled[0]):
        # Identical values everywhere: no difference.
        return True

    ranks = _average_ranks(pooled)
    N = n + m
    observed = abs(ranks[:n].mean() - ranks[n:].mean())

    rng = np.random.default_rng(0)
    total = 10000
    chunk = 1000
    extreme = 0
    done = 0
    tol = 1e-12
    while done < total:
        size = min(chunk, total - done)
        perm = np.argsort(rng.random((size, N)), axis=1)
        R = ranks[perm]
        diffs = np.abs(R[:, :n].mean(axis=1) - R[:, n:].mean(axis=1))
        extreme += int(np.sum(diffs >= observed - tol))
        done += size

    p_value = (extreme + 1) / (total + 1)
    return bool(p_value >= 0.05)
```
