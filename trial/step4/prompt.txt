  "-" goals to minimise; names ending in "X" are ignored; every other column is an input.

- Row indices count data rows of a csv from 0 (header excluded).

### step1.py
```python
import csv

import numpy as np
from sklearn.cluster import KMeans


def _read(path):
    with open(path, newline="") as f:
        reader = csv.DictReader(f)
        fieldnames = list(reader.fieldnames or [])
        rows = list(reader)
    return fieldnames, rows


def _columns(fieldnames):
    # goals: (name, sign) with sign +1 to maximise, -1 to minimise
    goals = [(c, 1.0) for c in fieldnames if c.endswith("+")]
    goals += [(c, -1.0) for c in fieldnames if c.endswith("-")]
    inputs = [c for c in fieldnames if not c.endswith(("+", "-", "X"))]
    return inputs, goals


def _to_float_column(rows, col):
    """Convert a column to floats; non-numeric strings are coded by first appearance."""
    values = [r[col] for r in rows]
    try:
        return np.array([float(v) for v in values], dtype=float)
    except (TypeError, ValueError):
        codes = {}
        out = np.empty(len(values), dtype=float)
        for i, v in enumerate(values):
            if v not in codes:
                codes[v] = float(len(codes))
            out[i] = codes[v]
        return out


def families(train_csv) -> list[int]:
    fieldnames, rows = _read(train_csv)
    n = len(rows)
    if n == 0:
        return []
    inputs, _ = _columns(fieldnames)
    if not inputs:
        return [0] * n
    X = np.column_stack([_to_float_column(rows, c) for c in inputs])
    k = min(5, n)
    km = KMeans(n_clusters=k, n_init=10, random_state=0)
    labels = km.fit_predict(X)
    return [int(x) for x in labels]


def best(train_csv) -> list[int]:
    fieldnames, rows = _read(train_csv)
    n = len(rows)
    _, goals = _columns(fieldnames)
    cap = int(np.floor(0.05 * n))
    if n == 0 or cap == 0 or not goals:
        return []

    # Oriented goal matrix: larger is better on every column.
    S = np.column_stack(
        [sign * _to_float_column(rows, name) for name, sign in goals]
    )

    # Pareto front: a row is kept unless some other row is at least as good
    # on every goal and strictly better on at least one.
    nondominated = np.ones(n, dtype=bool)
    for i in range(n):
        at_least = np.all(S >= S[i], axis=1)
        strictly = np.any(S > S[i], axis=1)
        if np.any(at_least & strictly):
            nondominated[i] = False
    front = np.flatnonzero(nondominated)

    if len(front) > cap:
        # Rank front rows by min-max normalised goal sum and keep the top `cap`.
        F = S[front]
        lo = S.min(axis=0)
        hi = S.max(axis=0)
        span = np.where(hi - lo > 0, hi - lo, 1.0)
        score = ((F - lo) / span).sum(axis=1)
        order = np.argsort(-score, kind="stable")
        front = front[order[:cap]]

    return sorted(int(i) for i in front)
```

```python
import numpy as np
from sklearn.neighbors import NearestNeighbors

from step1 import _read, _columns, _to_float_column


def _encode_inputs(rows, inputs):
    """Build per-column encoders from training rows, returning (matrix, encode_row)."""
    specs = []  # per input column: ("num", None) or ("cat", mapping dict)
    cols = []
    for c in inputs:
        values = [r[c] for r in rows]
        try:
            col = np.array([float(v) for v in values], dtype=float)
            specs.append(("num", None))
        except (TypeError, ValueError):
            mapping = {}
            for v in values:
                if v not in mapping:
                    mapping[v] = float(len(mapping))
            col = np.array([mapping[v] for v in values], dtype=float)
            specs.append(("cat", mapping))
        cols.append(col)

    X = np.column_stack(cols) if cols else np.zeros((len(rows), 0))
    col_means = X.mean(axis=0) if X.shape[1] else np.zeros(0)

    def encode_row(row_vals):
        out = np.empty(len(inputs), dtype=float)
        for j, (c, (kind, mapping)) in enumerate(zip(inputs, specs)):
            v = row_vals.get(c)
            if kind == "num":
                try:
                    out[j] = float(v)
                except (TypeError, ValueError):
                    out[j] = col_means[j]
                if not np.isfinite(out[j]):
                    out[j] = col_means[j]
            else:
                if v in mapping:
                    out[j] = mapping[v]
                else:
                    # Unseen category: fall back to the training mean code.
                    out[j] = col_means[j]
        return out

    return X, encode_row


def guess(train_csv, row) -> dict[str, tuple[float, float, float]]:
    fieldnames, rows = _read(train_csv)
    inputs, goals = _columns(fieldnames)
    n = len(rows)

    if n == 0:
        return {name: (float("nan"), float("-inf"), float("inf"))
                for name, _ in goals}

    X, encode_row = _encode_inputs(rows, inputs)

    # Standardise inputs so every feature contributes comparably to distance.
    if X.shape[1]:
        mu = X.mean(axis=0)
        sd = X.std(axis=0)
        sd = np.where(sd > 0, sd, 1.0)
        Z = (X - mu) / sd
        z = (encode_row(row) - mu) / sd
    else:
        Z = np.zeros((n, 1))
        z = np.zeros(1)

    k = min(15, n)
    nn = NearestNeighbors(n_neighbors=k).fit(Z)
    _, idx = nn.kneighbors(z.reshape(1, -1))
    idx = idx[0]

    result: dict[str, tuple[float, float, float]] = {}
    for name, _sign in goals:
        Y = _to_float_column(rows, name)
        neighbours = Y[idx]
        estimate = float(np.median(neighbours))
        low = float(np.percentile(neighbours, 10))
        high = float(np.percentile(neighbours, 90))
        result[name] = (estimate, low, high)

    return result
```

### step3.py
```python
import numpy as np

_ALPHA = 0.05
_N_PERM = 10000
_SEED = 0
_CHUNK = 500


def same(xs, ys) -> bool:
    """Return True if the gap between xs and ys is consistent with noise.

    Uses a two-sided permutation test on the difference in means. Non-finite
    values are dropped. With an empty sample or an observed difference of
    exactly zero, the samples are treated as not really different.
    """
    x = np.asarray([float(v) for v in xs], dtype=float)
    y = np.asarray([float(v) for v in ys], dtype=float)
    x = x[np.isfinite(x)]
    y = y[np.isfinite(y)]

    n, m = len(x), len(y)
    if n == 0 or m == 0:
        return True

    observed = abs(float(x.mean()) - float(y.mean()))
    if observed == 0.0:
        return True

    pooled = np.concatenate([x, y])
    total = n + m
    rng = np.random.default_rng(_SEED)

    threshold = observed * (1.0 - 1e-12)
    count = 0
    done = 0
    while done < _N_PERM:
        b = min(_CHUNK, _N_PERM - done)
        perm = rng.random((b, total)).argsort(axis=1)
        vals = pooled[perm]
        diff = np.abs(vals[:, :n].mean(axis=1) - vals[:, n:].mean(axis=1))
        count += int(np.count_nonzero(diff >= threshold))
        done += b

    p_value = (count + 1) / (_N_PERM + 1)
    return bool(p_value > _ALPHA)
```
